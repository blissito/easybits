import { useEffect, useRef, useState } from "react";
import { Link, createFileRoute, redirect, useRouter } from "@tanstack/react-router";
import { useServerFn } from "@tanstack/react-start";
import { getUser } from "../lib/session";
import { authClient } from "../lib/auth-client";
import { deleteChat, getAgentSettings, getWorkspace, saveAgentSettings, sendMessage } from "../lib/agent";

const SUGGESTIONS = [
  "Redacta un correo para pedir una reunión con un cliente nuevo",
  "Hazme un plan de contenido de una semana para Instagram",
  "Resume en 5 puntos las ventajas de trabajar por objetivos",
  "Ayúdame a preparar una entrevista de trabajo",
];

export const Route = createFileRoute("/app")({
  validateSearch: (s: Record<string, unknown>): { c?: string } => (typeof s.c === "string" ? { c: s.c } : {}),
  beforeLoad: async () => {
    if (!(await getUser())) throw redirect({ to: "/login" });
  },
  loaderDeps: ({ search }) => ({ c: search.c }),
  loader: ({ deps }) => getWorkspace({ data: { chatId: deps.c } }),
  head: () => ({ meta: [{ title: "Atlas · Chat" }] }),
  component: AgentApp,
});

type Msg = { id: string; role: "user" | "assistant"; content: string };

function AgentApp() {
  const ws = Route.useLoaderData();
  const router = useRouter();
  const send = useServerFn(sendMessage);
  const remove = useServerFn(deleteChat);
  const [pending, setPending] = useState<string | null>(null);
  const [error, setError] = useState("");
  const [text, setText] = useState("");
  const [menu, setMenu] = useState(false);
  const [settings, setSettings] = useState(false);
  const bottom = useRef<HTMLDivElement>(null);

  const messages: Msg[] = pending ? [...ws.messages, { id: "pending", role: "user", content: pending }] : ws.messages;

  useEffect(() => {
    bottom.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages.length, pending]);

  async function submit(value: string) {
    const msg = value.trim();
    if (!msg || pending) return;
    setError("");
    setText("");
    setPending(msg);
    try {
      const r = await send({ data: { chatId: ws.active?.id, text: msg } });
      if (r.chatId !== ws.active?.id) await router.navigate({ to: "/app", search: { c: r.chatId } });
      else await router.invalidate();
    } catch (e) {
      setText(msg);
      setError(e instanceof Error ? e.message : "No se pudo enviar");
    } finally {
      setPending(null);
    }
  }

  async function onDelete(id: string) {
    if (!confirm("¿Borrar esta conversación? No se puede deshacer.")) return;
    await remove({ data: { chatId: id } });
    if (id === ws.active?.id) await router.navigate({ to: "/app", search: { c: undefined } });
    else await router.invalidate();
  }

  async function signOut() {
    await authClient.signOut();
    await router.navigate({ to: "/" });
  }

  return (
    <div className="h-dvh flex bg-surface text-on-surface">
      <aside
        data-eb-id="sidebar"
        className={`${menu ? "flex" : "hidden"} md:flex fixed md:static inset-0 z-20 w-full md:w-72 flex-col bg-surface-alt p-4 gap-3`}
      >
        <div className="flex items-center gap-2">
          <Link to="/" className="mr-auto flex items-center gap-2 font-bold text-lg">
            <span className="grid place-items-center size-8 rounded-xl bg-primary text-on-primary">◆</span> {ws.agent.name}
          </Link>
          <button onClick={() => setMenu(false)} className="md:hidden text-2xl px-2" aria-label="Cerrar menú">×</button>
        </div>
        <Link
          to="/app"
          search={{ c: undefined }}
          onClick={() => setMenu(false)}
          className="rounded-xl bg-primary text-on-primary px-4 py-2.5 font-bold text-center hover:bg-primary-dark"
        >
          + Nueva conversación
        </Link>
        <nav className="flex-1 overflow-y-auto -mx-1 space-y-0.5">
          {ws.chats.length === 0 && <p className="px-2 py-4 text-sm text-on-surface-muted">Aún no tienes conversaciones.</p>}
          {ws.chats.map((c) => (
            <div key={c.id} className={`group flex items-center rounded-lg ${c.id === ws.active?.id ? "bg-primary-light" : "hover:bg-surface"}`}>
              <Link
                to="/app"
                search={{ c: c.id }}
                onClick={() => setMenu(false)}
                className="flex-1 truncate px-3 py-2 text-sm"
                title={c.title}
              >
                {c.title}
              </Link>
              <button
                onClick={() => onDelete(c.id)}
                className="px-2 text-on-surface-muted opacity-60 md:opacity-0 group-hover:opacity-100"
                aria-label="Borrar conversación"
              >
                🗑
              </button>
            </div>
          ))}
        </nav>
        <div className="border-t border-primary-light pt-3 text-sm space-y-2">
          {ws.user.isAdmin && (
            <button onClick={() => setSettings(true)} className="w-full text-left rounded-lg px-3 py-2 hover:bg-surface font-semibold">
              ⚙️ Ajustes del agente
            </button>
          )}
          <div className="flex items-center gap-2 px-3">
            <span className="grid place-items-center size-8 rounded-full bg-secondary text-on-secondary font-bold uppercase">{ws.user.name.slice(0, 1)}</span>
            <span className="flex-1 truncate">{ws.user.name}</span>
            <button onClick={signOut} className="text-on-surface-muted hover:text-on-surface">Salir</button>
          </div>
        </div>
      </aside>

      <main data-eb-id="chat" className="flex-1 flex flex-col min-w-0">
        <header className="flex items-center gap-3 px-4 md:px-8 py-3 border-b border-surface-alt">
          <button onClick={() => setMenu(true)} className="md:hidden text-xl" aria-label="Abrir menú">☰</button>
          <h1 className="font-bold truncate">{ws.active?.title ?? "Nueva conversación"}</h1>
        </header>

        {!ws.agent.ready && (
          <div data-eb-id="setup-banner" className="mx-4 md:mx-8 mt-4 rounded-xl bg-primary-light text-primary-dark px-4 py-3 text-sm">
            {ws.user.isAdmin ? (
              <>
                Falta conectar la IA para que {ws.agent.name} responda.{" "}
                <button onClick={() => setSettings(true)} className="font-bold underline">Agregar la llave</button>
              </>
            ) : (
              <>El agente se está configurando; pronto podrá responderte.</>
            )}
          </div>
        )}

        <div className="flex-1 overflow-y-auto">
          {messages.length === 0 ? (
            <div data-eb-id="empty" className="max-w-2xl mx-auto px-4 py-16 md:py-24">
              <p className="grid place-items-center size-14 rounded-2xl bg-primary text-on-primary text-2xl">◆</p>
              <h2 className="mt-5 text-3xl font-bold tracking-tight">Hola, {ws.user.name.split(" ")[0]}. ¿En qué te ayudo hoy?</h2>
              <div className="mt-8 grid sm:grid-cols-2 gap-3">
                {SUGGESTIONS.map((s) => (
                  <button key={s} onClick={() => submit(s)} className="text-left rounded-2xl bg-surface-alt p-4 text-sm hover:bg-primary-light">
                    {s}
                  </button>
                ))}
              </div>
            </div>
          ) : (
            <div className="max-w-3xl mx-auto px-4 md:px-8 py-6 space-y-5">
              {messages.map((m) =>
                m.role === "user" ? (
                  <p key={m.id} className="ml-auto w-fit max-w-[85%] whitespace-pre-wrap rounded-2xl rounded-br-md bg-primary text-on-primary px-4 py-3">
                    {m.content}
                  </p>
                ) : (
                  <div key={m.id} className="flex gap-3">
                    <span className="shrink-0 grid place-items-center size-8 rounded-lg bg-primary text-on-primary text-xs">◆</span>
                    <div className="min-w-0 whitespace-pre-wrap leading-relaxed pt-1">{m.content}</div>
                  </div>
                ),
              )}
              {pending && (
                <div className="flex gap-3 items-center text-on-surface-muted">
                  <span className="shrink-0 grid place-items-center size-8 rounded-lg bg-primary text-on-primary text-xs animate-pulse">◆</span>
                  {ws.agent.name} está pensando…
                </div>
              )}
              <div ref={bottom} />
            </div>
          )}
        </div>

        <form
          data-eb-id="composer"
          onSubmit={(e) => {
            e.preventDefault();
            submit(text);
          }}
          className="px-4 md:px-8 pb-5"
        >
          {error && <p className="max-w-3xl mx-auto mb-2 text-sm text-primary-dark">{error}</p>}
          <div className="max-w-3xl mx-auto flex items-end gap-2 rounded-3xl bg-surface-alt p-2 pl-5 focus-within:ring-2 focus-within:ring-primary">
            <textarea
              value={text}
              onChange={(e) => setText(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === "Enter" && !e.shiftKey) {
                  e.preventDefault();
                  submit(text);
                }
              }}
              rows={1}
              placeholder={`Escríbele a ${ws.agent.name}…`}
              className="flex-1 resize-none bg-transparent py-2.5 outline-none max-h-40"
            />
            <button
              disabled={!text.trim() || Boolean(pending)}
              className="grid place-items-center size-10 rounded-full bg-primary text-on-primary font-bold disabled:opacity-40"
              aria-label="Enviar"
            >
              ↑
            </button>
          </div>
          <p className="max-w-3xl mx-auto mt-2 text-center text-xs text-on-surface-muted">
            {ws.agent.name} puede equivocarse. Revisa la información importante.
          </p>
        </form>
      </main>

      {settings && <SettingsDialog onClose={() => setSettings(false)} />}
    </div>
  );
}

function SettingsDialog({ onClose }: { onClose: () => void }) {
  const router = useRouter();
  const load = useServerFn(getAgentSettings);
  const save = useServerFn(saveAgentSettings);
  const [s, setS] = useState<Awaited<ReturnType<typeof getAgentSettings>> | null>(null);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    load().then(setS, (e) => setError(String(e?.message ?? e)));
  }, []);

  async function submit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const f = new FormData(e.currentTarget);
    setBusy(true);
    setError("");
    try {
      await save({ data: { name: String(f.get("name")), prompt: String(f.get("prompt")), apiKey: String(f.get("apiKey") || "") } });
      await router.invalidate();
      onClose();
    } catch (err) {
      setError(err instanceof Error ? err.message : "No se pudo guardar");
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="fixed inset-0 z-30 grid place-items-center bg-secondary/60 p-4" onClick={onClose}>
      <form
        data-eb-id="settings"
        onSubmit={submit}
        onClick={(e) => e.stopPropagation()}
        className="w-full max-w-lg rounded-3xl bg-surface p-6 space-y-4 max-h-[90dvh] overflow-y-auto"
      >
        <div className="flex items-center">
          <h2 className="mr-auto text-xl font-bold">Ajustes del agente</h2>
          <button type="button" onClick={onClose} className="text-2xl px-2" aria-label="Cerrar">×</button>
        </div>
        {!s && !error && <p className="text-on-surface-muted">Cargando…</p>}
        {s && (
          <>
            <label className="block text-sm font-semibold">
              Nombre del agente
              <input name="name" defaultValue={s.name} required maxLength={40} className="mt-1 w-full rounded-xl border border-surface-alt bg-surface-alt p-3 font-normal" />
            </label>
            <label className="block text-sm font-semibold">
              Personalidad e instrucciones
              <textarea name="prompt" defaultValue={s.prompt} rows={6} className="mt-1 w-full rounded-xl border border-surface-alt bg-surface-alt p-3 font-normal" />
              <span className="font-normal text-on-surface-muted">Cómo habla, de qué sabe y qué debe evitar. Aplica a todos los usuarios.</span>
            </label>
            <label className="block text-sm font-semibold">
              Llave de IA (Anthropic)
              <input
                name="apiKey"
                type="password"
                autoComplete="off"
                placeholder={s.hasKey ? `Guardada (${s.keyHint}) · deja vacío para conservarla` : "sk-ant-…"}
                className="mt-1 w-full rounded-xl border border-surface-alt bg-surface-alt p-3 font-normal"
              />
              <span className="font-normal text-on-surface-muted">Sácala en console.anthropic.com → API Keys. Sólo tú la ves.</span>
            </label>
          </>
        )}
        {error && <p className="text-sm text-primary-dark">{error}</p>}
        <button disabled={!s || busy} className="w-full rounded-xl bg-primary text-on-primary p-3 font-bold disabled:opacity-50">
          {busy ? "Guardando…" : "Guardar"}
        </button>
      </form>
    </div>
  );
}
