// Conversaciones del agente: cada usuario ve sólo las suyas. El admin configura
// nombre, personalidad y la llave de Anthropic desde la propia app (tabla settings).
import { createServerFn } from "@tanstack/react-start";
import { getRequestHeaders } from "@tanstack/react-start/server";
import { auth, authReady } from "./auth";
import { db } from "./db";

const MODEL = "claude-sonnet-5-5";
const DEFAULT_NAME = "Atlas";
const DEFAULT_PROMPT =
  "Eres Atlas, un asistente de IA claro, cálido y directo. Respondes en el idioma de la persona, vas al grano y propones siguientes pasos concretos cuando ayudan.";

let schema: Promise<unknown> | undefined;
function ensureSchema() {
  return (schema ??= db
    .batch(
      [
        `CREATE TABLE IF NOT EXISTS chats (id TEXT PRIMARY KEY, user_id TEXT NOT NULL, title TEXT NOT NULL, created_at INTEGER NOT NULL, updated_at INTEGER NOT NULL)`,
        `CREATE INDEX IF NOT EXISTS chats_user ON chats(user_id, updated_at)`,
        `CREATE TABLE IF NOT EXISTS messages (id TEXT PRIMARY KEY, chat_id TEXT NOT NULL, role TEXT NOT NULL, content TEXT NOT NULL, created_at INTEGER NOT NULL)`,
        `CREATE INDEX IF NOT EXISTS messages_chat ON messages(chat_id, created_at)`,
        `CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT NOT NULL)`,
      ],
      "write",
    )
    .catch((e) => {
      schema = undefined;
      throw e;
    }));
}

async function currentUser() {
  await authReady();
  const s = await auth.api.getSession({ headers: getRequestHeaders() });
  if (!s) return null;
  const u = s.user as typeof s.user & { role?: string | null };
  return { id: u.id, name: u.name, email: u.email, isAdmin: u.role === "admin" };
}

async function requireUser() {
  const u = await currentUser();
  if (!u) throw new Error("Necesitas entrar a tu cuenta");
  await ensureSchema();
  return u;
}

async function readSettings() {
  const r = await db.execute("SELECT key, value FROM settings");
  const m = Object.fromEntries(r.rows.map((row) => [String(row.key), String(row.value)]));
  return {
    name: m.name || DEFAULT_NAME,
    prompt: m.prompt || DEFAULT_PROMPT,
    apiKey: m.api_key || process.env.ANTHROPIC_API_KEY || "",
  };
}

async function ownChat(chatId: string, userId: string) {
  const r = await db.execute({ sql: "SELECT id, title FROM chats WHERE id = ? AND user_id = ?", args: [chatId, userId] });
  if (!r.rows[0]) throw new Error("Conversación no encontrada");
  return { id: String(r.rows[0].id), title: String(r.rows[0].title) };
}

export const getMe = createServerFn({ method: "GET" }).handler(() => currentUser());

export const getWorkspace = createServerFn({ method: "GET" })
  .validator((d: { chatId?: string }) => d)
  .handler(async ({ data }) => {
    const user = await requireUser();
    const settings = await readSettings();
    const chats = await db.execute({
      sql: "SELECT id, title, updated_at FROM chats WHERE user_id = ? ORDER BY updated_at DESC LIMIT 100",
      args: [user.id],
    });
    let active: { id: string; title: string } | null = null;
    let messages: { id: string; role: "user" | "assistant"; content: string }[] = [];
    if (data.chatId) {
      try {
        active = await ownChat(data.chatId, user.id);
        const m = await db.execute({
          sql: "SELECT id, role, content FROM messages WHERE chat_id = ? ORDER BY created_at, rowid",
          args: [active.id],
        });
        messages = m.rows.map((r) => ({ id: String(r.id), role: r.role === "user" ? "user" : "assistant", content: String(r.content) }));
      } catch {
        active = null;
      }
    }
    return {
      user,
      agent: { name: settings.name, ready: Boolean(settings.apiKey) },
      chats: chats.rows.map((r) => ({ id: String(r.id), title: String(r.title), updatedAt: Number(r.updated_at) })),
      active,
      messages,
    };
  });

export const sendMessage = createServerFn({ method: "POST" })
  .validator((d: { chatId?: string; text: string }) => {
    const text = String(d.text ?? "").trim();
    if (!text) throw new Error("Escribe un mensaje");
    if (text.length > 8000) throw new Error("El mensaje es demasiado largo");
    return { chatId: d.chatId ? String(d.chatId) : undefined, text };
  })
  .handler(async ({ data }) => {
    const user = await requireUser();
    const now = Date.now();
    let chatId = data.chatId;
    if (chatId) await ownChat(chatId, user.id);
    else {
      chatId = crypto.randomUUID();
      const title = data.text.replace(/\s+/g, " ").slice(0, 60);
      await db.execute({
        sql: "INSERT INTO chats (id, user_id, title, created_at, updated_at) VALUES (?, ?, ?, ?, ?)",
        args: [chatId, user.id, title, now, now],
      });
    }
    await db.execute({
      sql: "INSERT INTO messages (id, chat_id, role, content, created_at) VALUES (?, ?, 'user', ?, ?)",
      args: [crypto.randomUUID(), chatId, data.text, now],
    });

    const history = await db.execute({
      sql: "SELECT role, content FROM messages WHERE chat_id = ? ORDER BY created_at, rowid",
      args: [chatId],
    });
    const settings = await readSettings();
    const reply = await askModel(
      settings,
      user.name,
      history.rows.slice(-40).map((r) => ({ role: r.role === "user" ? "user" : "assistant", content: String(r.content) })),
    );

    await db.batch(
      [
        {
          sql: "INSERT INTO messages (id, chat_id, role, content, created_at) VALUES (?, ?, 'assistant', ?, ?)",
          args: [crypto.randomUUID(), chatId, reply, Date.now()],
        },
        { sql: "UPDATE chats SET updated_at = ? WHERE id = ?", args: [Date.now(), chatId] },
      ],
      "write",
    );
    return { chatId };
  });

async function askModel(
  settings: Awaited<ReturnType<typeof readSettings>>,
  userName: string,
  messages: { role: "user" | "assistant"; content: string }[],
) {
  if (!settings.apiKey)
    return `Hola, soy ${settings.name}. Todavía no tengo conectado mi cerebro de IA: la persona que administra esta app debe agregar la llave en «Ajustes del agente». En cuanto lo haga, podré responderte de verdad.`;
  try {
    const res = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: { "content-type": "application/json", "x-api-key": settings.apiKey, "anthropic-version": "2023-06-01" },
      body: JSON.stringify({
        model: MODEL,
        max_tokens: 2048,
        system: `${settings.prompt}\n\nTu nombre es ${settings.name}. Hablas con ${userName}. Hoy es ${new Date().toLocaleDateString("es-MX", { dateStyle: "full", timeZone: "America/Mexico_City" })}.`,
        messages,
      }),
    });
    const json = (await res.json()) as { content?: { type: string; text?: string }[]; error?: { message?: string } };
    if (!res.ok) {
      console.error("[agent] Anthropic", res.status, json.error?.message);
      return res.status === 401
        ? "La llave de IA configurada no es válida. Pide a quien administra la app que la revise en «Ajustes del agente»."
        : "Tuve un problema para pensar la respuesta. Intenta de nuevo en un momento.";
    }
    return json.content?.filter((b) => b.type === "text").map((b) => b.text).join("\n").trim() || "…";
  } catch (e) {
    console.error("[agent]", e);
    return "No pude conectarme con el servicio de IA. Intenta de nuevo en un momento.";
  }
}

export const deleteChat = createServerFn({ method: "POST" })
  .validator((d: { chatId: string }) => ({ chatId: String(d.chatId) }))
  .handler(async ({ data }) => {
    const user = await requireUser();
    await ownChat(data.chatId, user.id);
    await db.batch(
      [
        { sql: "DELETE FROM messages WHERE chat_id = ?", args: [data.chatId] },
        { sql: "DELETE FROM chats WHERE id = ?", args: [data.chatId] },
      ],
      "write",
    );
    return { ok: true };
  });

export const getAgentSettings = createServerFn({ method: "GET" }).handler(async () => {
  const user = await requireUser();
  if (!user.isAdmin) throw new Error("Sólo quien administra la app puede ver esto");
  const s = await readSettings();
  return { name: s.name, prompt: s.prompt, hasKey: Boolean(s.apiKey), keyHint: s.apiKey ? `…${s.apiKey.slice(-4)}` : "" };
});

export const saveAgentSettings = createServerFn({ method: "POST" })
  .validator((d: { name: string; prompt: string; apiKey?: string }) => ({
    name: String(d.name ?? "").trim().slice(0, 40),
    prompt: String(d.prompt ?? "").trim().slice(0, 6000),
    apiKey: d.apiKey ? String(d.apiKey).trim() : "",
  }))
  .handler(async ({ data }) => {
    const user = await requireUser();
    if (!user.isAdmin) throw new Error("Sólo quien administra la app puede cambiar esto");
    const rows: [string, string][] = [
      ["name", data.name],
      ["prompt", data.prompt],
    ];
    if (data.apiKey) rows.push(["api_key", data.apiKey]);
    await db.batch(
      rows.map(([k, v]) => ({ sql: "INSERT INTO settings (key, value) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value = excluded.value", args: [k, v] })),
      "write",
    );
    return { ok: true };
  });
