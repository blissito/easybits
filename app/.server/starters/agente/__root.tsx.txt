import type { ReactNode } from "react";
import { Outlet, createRootRoute, HeadContent, Scripts } from "@tanstack/react-router";
import css from "../styles.css?url";

// Selector de elementos del editor de EasyBits: sólo carga dentro de su vista previa
// (?eb_edit=1 en un iframe). En producción no hace nada.
const EB_PICKER = `if(location.search.includes("eb_edit=1")&&window.top!==window){var s=document.createElement("script");s.src="https://www.easybits.cloud/eb-picker.js";document.body.appendChild(s)}`;

export const Route = createRootRoute({
  head: () => ({
    meta: [
      { charSet: "utf-8" },
      { name: "viewport", content: "width=device-width, initial-scale=1" },
      { title: "Atlas" },
    ],
    links: [{ rel: "stylesheet", href: css }],
  }),
  component: () => (
    <RootDocument>
      <Outlet />
    </RootDocument>
  ),
});

function RootDocument({ children }: Readonly<{ children: ReactNode }>) {
  return (
    <html lang="es">
      <head>
        <HeadContent />
      </head>
      <body>
        {children}
        <Scripts />
        <script dangerouslySetInnerHTML={{ __html: EB_PICKER }} />
      </body>
    </html>
  );
}
