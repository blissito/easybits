import { useState } from "react";
import { createFileRoute, useRouter } from "@tanstack/react-router";
import { authClient } from "../lib/auth-client";

export const Route = createFileRoute("/login")({ component: Login });

function Login() {
  const router = useRouter();
  const [mode, setMode] = useState<"in" | "up">("in");
  const [error, setError] = useState("");

  async function submit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError("");
    const f = new FormData(e.currentTarget);
    const email = String(f.get("email"));
    const password = String(f.get("password"));
    const r =
      mode === "up"
        ? await authClient.signUp.email({ email, password, name: String(f.get("name") || email) })
        : await authClient.signIn.email({ email, password });
    if (r.error) return setError(r.error.message ?? "No se pudo entrar");
    await router.navigate({ to: "/app" });
  }

  return (
    <main data-eb-id="login" className="min-h-screen grid place-items-center bg-surface p-8">
      <form data-eb-id="login-form" onSubmit={submit} className="w-full max-w-sm space-y-3">
        <h1 data-eb-id="login-title" className="text-2xl font-bold text-on-surface">
          {mode === "up" ? "Crea tu cuenta en Atlas" : "Entra a Atlas"}
        </h1>
        {mode === "up" && <input name="name" placeholder="Nombre" className="w-full rounded border p-2" />}
        <input name="email" type="email" required placeholder="Correo" className="w-full rounded border p-2" />
        <input name="password" type="password" required minLength={8} placeholder="Contraseña" className="w-full rounded border p-2" />
        {error && <p className="text-sm text-primary">{error}</p>}
        <button className="w-full rounded bg-primary p-2 text-on-primary">{mode === "up" ? "Crear cuenta" : "Entrar"}</button>
        <button type="button" onClick={() => setMode(mode === "up" ? "in" : "up")} className="w-full text-sm text-on-surface-muted">
          {mode === "up" ? "Ya tengo cuenta" : "Crear una cuenta"}
        </button>
      </form>
    </main>
  );
}
